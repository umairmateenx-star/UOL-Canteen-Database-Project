create database uol_canteen;
use uol_canteen;

-- 1. STRONG ENTITIES

create table student(
    student_id varchar(15) primary key,
    first_name varchar(40) not null,
    last_name varchar(40) not null,
    street varchar(80),
    city varchar(40),
    cnic varchar(15) unique,
    email varchar(80) unique not null,
    enrollment_date date not null,
    student_status varchar(15) check (student_status in ('Active', 'Suspended', 'Graduated'))
);

create table employee(
    emp_id varchar(10) primary key,
    first_name varchar(40) not null,
    last_name varchar(40) not null,
    emerg_name varchar(50),
    emerg_phone varchar(20),
    cnic varchar(15) unique,
    email varchar(80) unique,
    hire_date date not null,
    salary decimal(10,2) check (salary >= 0),
    emp_status varchar(15) check (emp_status in ('Active', 'On-leave', 'Resigned'))
);

create table stall(
    stall_id varchar(10) primary key,
    stall_name varchar(60) not null,
    block varchar(20),
    wing varchar(20),
    location_block varchar(30),
    floor_no int,
    opening_time time,
    closing_time time,
    is_active boolean default true,
    monthly_rent decimal(12,2)
);

create table vendor(
    vendor_id varchar(10) primary key,
    company_name varchar(100) not null,
    street varchar(80),
    city varchar(40),
    postal_code varchar(10),
    contact_name varchar(50),
    contact_phone varchar(20),
    contract_date date,
    ntn varchar(20) unique,
    is_approved boolean default false,
    license_expiry date
);

create table food_item(
    item_id varchar(12) primary key,
    item_name varchar(80) not null,
    calories int,
    protein_gram decimal(6,2),
    unit_price decimal(10,2) check (unit_price >= 0),
    category varchar(30),
    is_available boolean default true,
    prep_time int check (prep_time >= 0),
    last_restock_date date
);

create table canteen_order(
    order_id varchar(14) primary key,
    order_datetime datetime not null,
    order_type varchar(12) check (order_type in ('Dine-in', 'Takeaway')),
    order_channel varchar(12),
    table_no varchar(10),
    status varchar(12) check (status in ('Placed', 'Preparing', 'Served', 'Cancelled')),
    served_time datetime,
    payment_status varchar(12) check (payment_status in ('Paid', 'Unpaid', 'Refunded')),
    student_id varchar(15),
    emp_id varchar(10),
    stall_id varchar(10),
    foreign key (student_id) references student(student_id),
    foreign key (emp_id) references employee(emp_id),
    foreign key (stall_id) references stall(stall_id)
);

create table payment(
    payment_id varchar(14) primary key,
    pay_datetime datetime not null,
    amount decimal(12,2) check (amount > 0),
    pay_method varchar(8) check (pay_method in ('Cash', 'Card', 'Wallet')),
    pay_status varchar(12) check (pay_status in ('Completed', 'Pending', 'Failed')),
    cash_received decimal(12,2),
    coupon_code varchar(20),
    tip_amount decimal(10,2) default 0.00,
    order_id varchar(14) unique not null,
    student_id varchar(15),
    received_by_emp_id varchar(10),
    foreign key (order_id) references canteen_order(order_id),
    foreign key (student_id) references student(student_id),
    foreign key (received_by_emp_id) references employee(emp_id)
);

create table feedback(
    feedback_id varchar(14) primary key,
    student_id varchar(15),
    order_id varchar(14),
    item_id varchar(12),
    rating int check (rating between 1 and 5),
    comment varchar(300),
    feedback_datetime datetime,
    response_status varchar(12) check (response_status in ('Open', 'Replied', 'Closed')),
    response_text varchar(300),
    is_anonymous boolean default false,
    helpful_count int default 0,
    foreign key (student_id) references student(student_id),
    foreign key (order_id) references canteen_order(order_id),
    foreign key (item_id) references food_item(item_id)
);


-- 2. WEAK ENTITIES

create table menu(
    stall_id varchar(10),
    menu_id varchar(10),
    menu_date date not null,
    meal_type varchar(15) check (meal_type in ('Breakfast', 'Lunch', 'Snacks')),
    is_active boolean default true,
    publish_time datetime,
    menu_version int default 1,
    special_note varchar(160),
    approved_by varchar(10),
    primary key (stall_id, menu_id),
    foreign key (stall_id) references stall(stall_id),
    foreign key (approved_by) references employee(emp_id)
);

create table order_line(
    order_id varchar(14),
    line_no int,
    item_id varchar(12) not null,
    quantity int check (quantity > 0),
    unit_price decimal(10,2) check (unit_price >= 0),
    discount decimal(10,2) default 0.00,
    special_instructions varchar(160),
    serving_size varchar(12),
    kot_status varchar(15),
    preparation_note varchar(160),
    primary key (order_id, line_no),
    foreign key (order_id) references canteen_order(order_id),
    foreign key (item_id) references food_item(item_id)
);


-- 3. MULTIVALUED ATTRIBUTES

create table student_phone(
    student_id varchar(15),
    phone varchar(20),
    primary key (student_id, phone),
    foreign key (student_id) references student(student_id)
);

create table employee_phone(
    emp_id varchar(10),
    phone varchar(20),
    primary key (emp_id, phone),
    foreign key (emp_id) references employee(emp_id)
);

create table vendor_phone(
    vendor_id varchar(10),
    phone varchar(20),
    primary key (vendor_id, phone),
    foreign key (vendor_id) references vendor(vendor_id)
);

create table food_item_allergen(
    item_id varchar(12),
    allergen varchar(50),
    primary key (item_id, allergen),
    foreign key (item_id) references food_item(item_id)
);

create table stall_facility(
    stall_id varchar(10),
    facility varchar(50),
    primary key (stall_id, facility),
    foreign key (stall_id) references stall(stall_id)
);


-- 4. M:N RELATIONSHIPS

create table works_at(
    emp_id varchar(10),
    stall_id varchar(10),
    shift_hours int,
    primary key (emp_id, stall_id),
    foreign key (emp_id) references employee(emp_id),
    foreign key (stall_id) references stall(stall_id)
);

create table menu_item(
    stall_id varchar(10),
    menu_id varchar(10),
    item_id varchar(12),
    offered_price decimal(10,2) check (offered_price >= 0),
    primary key (stall_id, menu_id, item_id),
    foreign key (stall_id, menu_id) references menu(stall_id, menu_id),
    foreign key (item_id) references food_item(item_id)
);

create table supplies(
    vendor_id varchar(10),
    item_id varchar(12),
    supply_date date,
    supply_price decimal(10,2),
    primary key (vendor_id, item_id, supply_date),
    foreign key (vendor_id) references vendor(vendor_id),
    foreign key (item_id) references food_item(item_id)
);


-- 5. SUBCLASSES (EER Mapping)

create table manager(
    emp_id varchar(10) primary key,
    mgr_level varchar(20),
    approved_budget decimal(12,2),
    foreign key (emp_id) references employee(emp_id)
);

create table cook(
    emp_id varchar(10) primary key,
    specialty varchar(50),
    hygiene_cert_no varchar(20),
    cert_expiry date,
    foreign key (emp_id) references employee(emp_id)
);

create table cashier(
    emp_id varchar(10) primary key,
    counter_no int,
    drawer_code varchar(10),
    foreign key (emp_id) references employee(emp_id)
);

create table meal(
    item_id varchar(12) primary key,
    cuisine varchar(30),
    spice_level varchar(15),
    is_vegetarian boolean,
    foreign key (item_id) references food_item(item_id)
);

create table beverage(
    item_id varchar(12) primary key,
    size varchar(15),
    is_chilled boolean,
    caffeine_level varchar(15),
    foreign key (item_id) references food_item(item_id)
);

create table snack(
    item_id varchar(12) primary key,
    is_fried boolean,
    is_baked boolean,
    portion_weight_grams int,
    foreign key (item_id) references food_item(item_id)
);

create table cash_payment(
    payment_id varchar(14) primary key,
    received_by_emp_id varchar(10),
    cash_received decimal(12,2),
    foreign key (payment_id) references payment(payment_id),
    foreign key (received_by_emp_id) references employee(emp_id)
);

create table card_payment(
    payment_id varchar(14) primary key,
    card_no varchar(20),
    card_type varchar(15),
    auth_code varchar(15),
    bank_name varchar(50),
    foreign key (payment_id) references payment(payment_id)
);

create table wallet_payment(
    payment_id varchar(14) primary key,
    wallet_app varchar(30),
    txn_ref varchar(30),
    wallet_account varchar(30),
    provider_fee decimal(10,2),
    foreign key (payment_id) references payment(payment_id)
);

select o.order_id, o.order_type, s.first_name, s.last_name
from canteen_order o
inner join student s on o.student_id = s.student_id;

-- 2. INNER JOIN (Query 2): Show food items and their offered price on menus.
select f.item_name, m.offered_price
from food_item f
inner join menu_item m on f.item_id = m.item_id;

-- 3. LEFT JOIN (Query 1): Show all students and their orders (even if they made no orders).
select s.first_name, o.order_id
from student s
left join canteen_order o on s.student_id = o.student_id;

-- 4. LEFT JOIN (Query 2): Show all employees and the stalls they work at (even if they work nowhere).
select e.first_name, w.stall_id
from employee e
left join works_at w on e.emp_id = w.emp_id;

-- 5. RIGHT JOIN (Query 1): Show all orders and the student who placed them (even if student ID is missing).
select o.order_id, s.first_name
from student s
right join canteen_order o on s.student_id = o.student_id;

-- 6. RIGHT JOIN (Query 2): Show all food items and any orders that included them.
select ol.order_id, f.item_name
from order_line ol
right join food_item f on ol.item_id = f.item_id;

-- 7. FULL OUTER JOIN (Query 1): (MySQL doesn't support FULL OUTER, so we use UNION of LEFT and RIGHT).
select s.first_name, o.order_id
from student s
left join canteen_order o on s.student_id = o.student_id
union
select s.first_name, o.order_id
from student s
right join canteen_order o on s.student_id = o.student_id;

-- 8. FULL OUTER JOIN (Query 2): Show all payments and their matching cash payment details.
select p.payment_id, c.cash_received
from payment p
left join cash_payment c on p.payment_id = c.payment_id
union
select p.payment_id, c.cash_received
from payment p
right join cash_payment c on p.payment_id = c.payment_id;

-- 9. CROSS JOIN (Query 1): Show every possible combination of stalls and food items.
select s.stall_name, f.item_name
from stall s
cross join food_item f;

-- 10. CROSS JOIN (Query 2): Show every combination of employees and vendors.
select e.first_name, v.company_name
from employee e
cross join vendor v;

-- 11. SELF JOIN (Query 1): Show employees who earn more than another employee.
select a.first_name as emp1, b.first_name as emp2, a.salary, b.salary
from employee a
join employee b on a.salary > b.salary;

-- 12. SELF JOIN (Query 2): Show students who live in the same city.
select a.first_name as student1, b.first_name as student2, a.city
from student a
join student b on a.city = b.city and a.student_id < b.student_id;

select item_name, unit_price from food_item

-- 3. Multiple-row Subquery (IN): Find students who have placed at least one order.
select first_name from student

-- 4. Multiple-row Subquery (IN): Find food items that have never been ordered.
select item_name from food_item

select company_name from vendor

    select vendor_id from supplies 
    where item_id in (select item_id from food_item where item_name = 'Chicken Biryani')
);

select first_name from student s 
where exists (select 1 from feedback f where f.student_id = s.student_id);

-- 7. Subquery with NOT EXISTS: Find vendors who have never supplied any food items.
select company_name from vendor v 
where not exists (select 1 from supplies s where s.vendor_id = v.vendor_id);

-- 8. Correlated Subquery: Find food items more expensive than the average price in their own category.
select f1.item_name, f1.unit_price, f1.category 
from food_item f1
where unit_price > (select avg(unit_price) from food_item f2 where f2.category = f1.category);

-- 9. Correlated Subquery: Find employees who earn more than the average salary of all employees in their same status.
select e1.first_name, e1.salary, e1.emp_status from employee e1
where salary > (select avg(salary) from employee e2 where e2.emp_status = e1.emp_status);

    select student_id from payment

);
